-- Utility Functions
local function createCorner(radius)
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, radius)
    return corner
end

local function createStroke(color, thickness, transparency)
    local stroke = Instance.new("UIStroke")
    stroke.Color = color
    stroke.Thickness = thickness
    stroke.Transparency = transparency or 0
    return stroke
end

local function createGradient(colors, rotation)
    local gradient = Instance.new("UIGradient")
    gradient.Color = ColorSequence.new(colors)
    gradient.Rotation = rotation or 0
    return gradient
end

local function tween(obj, props, duration, style, direction)
    return TweenService:Create(
        obj,
        TweenInfo.new(duration or 0.2, style or Enum.EasingStyle.Quad, direction or Enum.EasingDirection.Out),
        props
    )
end

-- Core Functions
local isExecuting = false

local function updateStatus(text, color, indicatorColor)
    statusLabel.Text = text
    statusLabel.TextColor3 = color
    if indicatorColor then
        tween(statusIndicator, {BackgroundColor3 = indicatorColor}, 0.3):Play()
    end
end

local function validateScript(code)
    if code == "" or code == "-- Backdoor Executor\nprint('Hello World')" then
        return false, "No script entered"
    end
    if #code > MAX_SCRIPT_SIZE then
        return false, "Script too large (max " .. (MAX_SCRIPT_SIZE/1000) .. "k characters)"
    end
    return true, nil
end

local function validateRepetitions(text)
    if text == "" or text:match("^%s*$") then
        return true, 1
    end
    
    local num = tonumber(text)
    if not num then
        return false, "Invalid input - enter a number"
    end
    
    if num ~= math.floor(num) then
        return false, "Must be a whole number"
    end
    
    if num < 0 or num > MAX_REPETITIONS then
        return false, "Must be between 0 and " .. MAX_REPETITIONS
    end
    
    return true, math.floor(num)
end

local function executeScript(code)
    local remoteCount = 0
    local descendants = ReplicatedStorage:GetDescendants()
    
    for _, obj in ipairs(descendants) do
        if obj:IsA("RemoteEvent") then
            pcall(function()
                obj:FireServer(code)
                remoteCount += 1
            end)
        end
    end
    
    return remoteCount
end

-- Execute Button Handler
executeBtn.MouseButton1Click:Connect(function()
    if isExecuting then
        updateStatus("⚠ Execution in progress, please wait...", COLORS.Warning, Color3.fromRGB(255, 200, 100))
        return
    end
    
    local code = scriptInput.Text
    local valid, error = validateScript(code)
    
    if not valid then
        updateStatus("Error: " .. error, COLORS.Error, Color3.fromRGB(255, 100, 100))
        task.wait(2)
        updateStatus("Ready to execute", Color3.fromRGB(200, 200, 210), Color3.fromRGB(100, 255, 150))
        return
    end
    
    local repValid, repCount = validateRepetitions(repInput.Text)
    
    if not repValid then
        updateStatus("Repetition Error: " .. repCount, COLORS.Error, Color3.fromRGB(255, 100, 100))
        tween(repInputStroke, {Color = Color3.fromRGB(255, 100, 100), Transparency = 0, Thickness = 3}, 0.15):Play()
        task.wait(0.3)
        tween(repInputStroke, {Color = Color3.fromRGB(70, 70, 85), Transparency = 0.3, Thickness = 2}, 0.3):Play()
        task.wait(1.5)
        updateStatus("Ready to execute", Color3.fromRGB(200, 200, 210), Color3.fromRGB(100, 255, 150))
        return
    end
    
    isExecuting = true
    
    if repCount == 0 then
        updateStatus("⚠ 0 repetitions - no execution performed", COLORS.Warning, Color3.fromRGB(255, 200, 100))
        isExecuting = false
        task.wait(2)
        updateStatus("Ready to execute", Color3.fromRGB(200, 200, 210), Color3.fromRGB(100, 255, 150))
        return
    end
    
    updateStatus("Executing script...", COLORS.Warning, Color3.fromRGB(255, 200, 100))
    tween(executeBtn, {BackgroundColor3 = Color3.fromRGB(80, 240, 160)}, 0.1):Play()
    task.wait(0.1)
    tween(executeBtn, {BackgroundColor3 = COLORS.ExecuteBtn}, 0.1):Play()
    
    if repCount == 1 then
        local remoteCount = executeScript(code)
        task.wait(0.3)
        
        if remoteCount > 0 then
            updateStatus("✓ Executed successfully! Fired " .. remoteCount .. " remotes", COLORS.Success, Color3.fromRGB(100, 255, 150))
        else
            updateStatus("⚠ No remotes found in ReplicatedStorage", COLORS.Warning, Color3.fromRGB(255, 200, 100))
        end
        
        isExecuting = false
        task.wait(3)
        updateStatus("Ready to execute", Color3.fromRGB(200, 200, 210), Color3.fromRGB(100, 255, 150))
    else
        task.spawn(function()
            local totalRemotes = 0
            local executionCount = 0
            
            for i = 1, repCount do
                if not isExecuting then break end
                
                totalRemotes += executeScript(code)
                executionCount = i
                
                if i % 20 == 0 or i == repCount or i == 1 or i == 10 or i == 50 or i == 100 then
                    updateStatus("⚡ Executing: " .. i .. "/" .. repCount, Color3.fromRGB(150, 200, 255), Color3.fromRGB(120, 170, 255))
                end
                
                if i % 50 == 0 then
                    task.wait()
                end
            end
            
            isExecuting = false
            
            if executionCount == repCount then
                updateStatus("✓ Complete! " .. executionCount .. " executions, " .. totalRemotes .. " total remotes fired", COLORS.Success, Color3.fromRGB(100, 255, 150))
            else
                updateStatus("⚠ Interrupted at " .. executionCount .. "/" .. repCount, COLORS.Warning, Color3.fromRGB(255, 200, 100))
            end
            
            task.wait(4)
            updateStatus("Ready to execute", Color3.fromRGB(200, 200, 210), Color3.fromRGB(100, 255, 150))
        end)
    end
end)

    local text = repInput.Text
    local filtered = text:gsub("[^0-9]", "")
    if filtered ~= text then
        repInput.Text = filtered
    end
    if #filtered > 3 then
        repInput.Text = filtered:sub(1, 3)
    end
end)
