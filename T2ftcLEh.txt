    statusLabel.Text = text
    statusLabel.TextColor3 = color

        TweenService:Create(statusIndicator, TweenInfo.new(0.3), {

local function validateScript(code)
    if code == "" or code == "-- Backdoor Executor\nprint('Hello World')" then
        return false, "No script entered"
    end
    if #code > 250000 then
        return false, "Script too large (max 250k characters)"
    end
    return true, nil
end

local function validateRepetitions(text)
    -- If empty or whitespace, default to 1
    if text == "" or text:match("^%s*$") then
        return true, 1
    end
    
    -- Check if it's a valid number
    local num = tonumber(text)
    
    -- Not a valid number
    if not num then
        return false, "Invalid input - enter a number"
    end
    
    -- Check if it's an integer
    if num ~= math.floor(num) then
        return false, "Must be a whole number"
    end
    
    -- Check range
    if num < 0 or num > 998 then
        return false, "Must be between 0 and 998"
    end
    
    return true, math.floor(num)
end

local function executeScript(code, isMultiple)
    local remoteCount = 0
    local replicatedStorage = game:GetService("ReplicatedStorage")
    
    for _, obj in pairs(replicatedStorage:GetDescendants()) do
        if obj:IsA("RemoteEvent") then
            local success, err = pcall(function()
                obj:FireServer(code)
                remoteCount = remoteCount + 1
            end)
            if not success and not isMultiple then
                warn("Error firing remote:", err)
            end
        end
    end
    
    return remoteCount
end

    end

        }):Play()

        return
    end
    
    isExecuting = true
    
    -- Handle 0 repetitions
    if repCount == 0 then
        updateStatus("⚠ 0 repetitions - no execution performed", Color3.fromRGB(255, 200, 120), Color3.fromRGB(255, 200, 100))
        isExecuting = false
        wait(2)
        updateStatus("Ready to execute", Color3.fromRGB(200, 200, 210), Color3.fromRGB(100, 255, 150))
        return
    end
    
    -- Single execution
    if repCount == 1 then

        TweenService:Create(executeBtn, TweenInfo.new(0.1), {BackgroundColor3 = Color3.fromRGB(80, 240, 160)}):Play()

        wait(0.3)
        
        if remoteCount > 0 then
            updateStatus("✓ Executed successfully! Fired " .. remoteCount .. " remotes", Color3.fromRGB(150, 255, 180), Color3.fromRGB(100, 255, 150))
        else
            updateStatus("⚠ No remotes found in ReplicatedStorage", Color3.fromRGB(255, 200, 120), Color3.fromRGB(255, 200, 100))
        end
        
        isExecuting = false
        wait(3)
        updateStatus("Ready to execute", Color3.fromRGB(200, 200, 210), Color3.fromRGB(100, 255, 150))
    
    -- Multiple executions
    else
        updateStatus("⚡ Executing: 0/" .. repCount, Color3.fromRGB(150, 200, 255), Color3.fromRGB(120, 170, 255))
        
        -- Button animation
        TweenService:Create(executeBtn, TweenInfo.new(0.1), {BackgroundColor3 = Color3.fromRGB(80, 240, 160)}):Play()
        wait(0.1)
        TweenService:Create(executeBtn, TweenInfo.new(0.1), {BackgroundColor3 = Color3.fromRGB(60, 200, 120)}):Play()
        
        spawn(function()
            local totalRemotes = 0
            local executionCount = 0
            
            for i = 1, repCount do
                if not isExecuting then break end
                
                local remoteCount = executeScript(code, true)
                totalRemotes = totalRemotes + remoteCount
                executionCount = i
                
                -- Update status at key milestones
                if i % 20 == 0 or i == repCount or i == 1 or i == 10 or i == 50 or i == 100 then
                    updateStatus("⚡ Executing: " .. i .. "/" .. repCount, Color3.fromRGB(150, 200, 255), Color3.fromRGB(120, 170, 255))
                end
                
                -- Small delay to prevent lockup
                if i % 50 == 0 then
                    task.wait()
                end
            end
            
            isExecuting = false
            
            if executionCount == repCount then
                updateStatus("✓ Complete! " .. executionCount .. " executions, " .. totalRemotes .. " total remotes fired", Color3.fromRGB(150, 255, 180), Color3.fromRGB(100, 255, 150))
            else
                updateStatus("⚠ Interrupted at " .. executionCount .. "/" .. repCount, Color3.fromRGB(255, 200, 120), Color3.fromRGB(255, 200, 100))
            end
            
            wait(4)
            updateStatus("Ready to execute", Color3.fromRGB(200, 200, 210), Color3.fromRGB(100, 255, 150))
        end)
    end
end)

    local text = repInput.Text

    end
